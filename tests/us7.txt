#####################################################################################
# User Story 7 - Consulta de disponibilidade (bicicletas / vagas)
#####################################################################################
# (refatorado) Este arquivo agora so depende das US1-US7: o trecho que usava
# retirada (US8) foi movido para o us8.txt e o que usava manutencao (US14)
# foi movido para o us14.txt.

zerarSistema

estacao1=criarEstacao nome="Estacao Universitaria" endereco="Av. Universitaria, 900" capacidade=4
expect 0 getBicicletasDisponiveis estacao=${estacao1}
expect 4 getVagasDisponiveis estacao=${estacao1}

adicionarBicicleta estacao=${estacao1} bicicleta=d1 tipo=convencional
adicionarBicicleta estacao=${estacao1} bicicleta=d2 tipo=convencional
adicionarBicicleta estacao=${estacao1} bicicleta=d3 tipo=convencional
expect 3 getBicicletasDisponiveis estacao=${estacao1}
expect 1 getVagasDisponiveis estacao=${estacao1}

# estacao cheia: nao ha mais vaga para uma quarta bicicleta alem da capacidade
adicionarBicicleta estacao=${estacao1} bicicleta=d4 tipo=convencional
expect 4 getBicicletasDisponiveis estacao=${estacao1}
expect 0 getVagasDisponiveis estacao=${estacao1}
expectError "Estacao sem vagas disponiveis." adicionarBicicleta estacao=${estacao1} bicicleta=d5 tipo=convencional

# ---- (refatorado) heranca/polimorfismo: eletrica abaixo do nivel minimo (15%) ocupa vaga
# ---- mas nao conta como disponivel; convencional nao tem essa restricao ----
estacao2=criarEstacao nome="Estacao Eletrica" endereco="Av. Eletrica, 15" capacidade=5
adicionarBicicleta estacao=${estacao2} bicicleta=bePouca tipo=eletrica nivelBateria=10
adicionarBicicleta estacao=${estacao2} bicicleta=beQuase tipo=eletrica nivelBateria=14
expect 0 getBicicletasDisponiveis estacao=${estacao2}
expect 3 getVagasDisponiveis estacao=${estacao2}

# fronteira: exatamente 15% ja conta como disponivel
adicionarBicicleta estacao=${estacao2} bicicleta=beLimite tipo=eletrica nivelBateria=15
expect 1 getBicicletasDisponiveis estacao=${estacao2}
adicionarBicicleta estacao=${estacao2} bicicleta=beMax tipo=eletrica nivelBateria=80
adicionarBicicleta estacao=${estacao2} bicicleta=bcNormal tipo=convencional
expect 3 getBicicletasDisponiveis estacao=${estacao2}
expect 0 getVagasDisponiveis estacao=${estacao2}

# ---- (refatorado) remover uma bicicleta atualiza as duas contagens ----
removerBicicleta estacao=${estacao1} bicicleta=d4
expect 3 getBicicletasDisponiveis estacao=${estacao1}
expect 1 getVagasDisponiveis estacao=${estacao1}

# ---- (refatorado) cada estacao tem a sua contagem ----
estacao3=criarEstacao nome="Estacao Pequena" endereco="Rua Curta, 1" capacidade=2
expect 0 getBicicletasDisponiveis estacao=${estacao3}
expect 2 getVagasDisponiveis estacao=${estacao3}
adicionarBicicleta estacao=${estacao3} bicicleta=d5 tipo=convencional
expect 1 getBicicletasDisponiveis estacao=${estacao3}
expect 1 getVagasDisponiveis estacao=${estacao3}
expect 3 getBicicletasDisponiveis estacao=${estacao1}
expect 3 getBicicletasDisponiveis estacao=${estacao2}

# ---- (refatorado) estado e tipo das bicicletas cadastradas ----
expect disponivel getEstadoBicicleta bicicleta=d1
expect eletrica getAtributoBicicleta bicicleta=beLimite atributo=tipo
expect ${estacao3} getAtributoBicicleta bicicleta=d5 atributo=estacao

encerrarSistema
quit
