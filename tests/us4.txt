#####################################################################################
# User Story 4 - Remocao de usuario
#####################################################################################

zerarSistema

id1=criarUsuario nome="Marcos Dias" email="marcos@exemplo.com" senha="123456" tipo=pessoaFisica cpf=300.400.500-60
id2=criarUsuario nome="Julia Reis" email="julia@exemplo.com" senha="654321" tipo=pessoaFisica cpf=600.500.400-30
id3=criarUsuario nome="Fernanda Melo" email="fernanda@exemplo.com" senha="112233" tipo=pessoaFisica cpf=700.600.500-40

# remocao simples
removerUsuario usuario=${id2}
expectError "Usuario nao existe." getAtributoUsuario usuario=${id2} atributo=nome
expectError "Usuario nao existe." removerUsuario usuario=${id2}

# erros gerais
expectError "Identificacao do usuario nao pode ser nula." removerUsuario usuario=""
expectError "Usuario nao existe." removerUsuario usuario=U999

# remocao de usuario sem historico
removerUsuario usuario=${id1}
expectError "Usuario nao existe." getAtributoUsuario usuario=${id1} atributo=nome

# (refatorado) a regra "nao remover usuario com aluguel em andamento" depende de retirada/devolucao
# e passou a ser testada em us8.txt e us9.txt

# (refatorado) o email de um usuario removido deixa de existir e pode ser reutilizado
expectError "Usuario nao existe." getUsuarioPorEmail email=marcos@exemplo.com
expectError "Usuario nao existe." getUsuarioPorEmail email=julia@exemplo.com
id4=criarUsuario nome="Julia Nova" email="julia@exemplo.com" senha="445566" tipo=pessoaFisica cpf=800.700.600-50
expectDifferent ${id2} echo ${id4}
expect ${id4} getUsuarioPorEmail email=julia@exemplo.com
expect "Julia Nova" getAtributoUsuario usuario=${id4} atributo=nome

# ---- (refatorado) identificacao so com espacos conta como vazia ----
expectError "Identificacao do usuario nao pode ser nula." removerUsuario usuario="   "

# ---- (refatorado) remover um usuario nao afeta os demais ----
expect "Fernanda Melo" getAtributoUsuario usuario=${id3} atributo=nome
expect fernanda@exemplo.com getAtributoUsuario usuario=${id3} atributo=email
expect ativo getAtributoUsuario usuario=${id3} atributo=estado
expect avulso getAtributoUsuario usuario=${id3} atributo=plano
expect ${id3} getUsuarioPorEmail email=fernanda@exemplo.com

# ---- (refatorado) usuario removido nao pode mais ser alterado ----
expectError "Usuario nao existe." alterarUsuario usuario=${id1} atributo=nome valor="Fantasma"
expectError "Usuario nao existe." alterarUsuario usuario=${id1} atributo=metodoPagamento valor1=pix chave=fantasma@pix.com
expectError "Usuario nao existe." alterarUsuario usuario=${id1} atributo=metodoPagamento valor1=cartao numero=1111 validade=12/2028

# ---- (refatorado) remocao de pessoa juridica assinante com metodo de pagamento ----
id5=criarUsuario nome="Pedala Corp" email="pedala@exemplo.com" senha="778899" \
	tipo=pessoaJuridica cnpj=88.888.888/0001-88 razaoSocial="Pedala Corporation" plano=assinante mensalidade=80
alterarUsuario usuario=${id5} atributo=metodoPagamento valor1=saldoPreCarregado
alterarUsuario usuario=${id5} atributo=saldo valor=30
removerUsuario usuario=${id5}
expectError "Usuario nao existe." getAtributoUsuario usuario=${id5} atributo=razaoSocial
expectError "Usuario nao existe." getUsuarioPorEmail email=pedala@exemplo.com

# ---- (refatorado) identificadores nao sao reaproveitados depois de remocoes ----
expect U5 echo ${id5}
id6=criarUsuario nome="Gil Prado" email="gil@exemplo.com" senha="123456" tipo=pessoaFisica cpf=111.000.222-33
expect U6 echo ${id6}
expectDifferent ${id5} echo ${id6}

encerrarSistema
quit
