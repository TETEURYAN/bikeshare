#####################################################################################
# User Story 2 - Consulta de usuario
#####################################################################################

zerarSistema

id1=criarUsuario nome="Joana Prado" email="joana@exemplo.com" senha="123456" tipo=pessoaFisica cpf=100.200.300-40
id2=criarUsuario nome="Loja Bikes ME" email="loja@exemplo.com" senha="654321" \
	tipo=pessoaJuridica cnpj=11.222.333/0001-44 razaoSocial="Loja Bikes Comercio ME"

# consulta basica de todos os atributos
expect "Joana Prado" getAtributoUsuario usuario=${id1} atributo=nome
expect joana@exemplo.com getAtributoUsuario usuario=${id1} atributo=email
expect 123456 getAtributoUsuario usuario=${id1} atributo=senha
expect ativo getAtributoUsuario usuario=${id1} atributo=estado
expect nenhum getAtributoUsuario usuario=${id1} atributo=aluguelAtivo
expect 0,00 getAtributoUsuario usuario=${id1} atributo=totalPendencias

# metodo de pagamento ainda nao cadastrado
expect nenhum getAtributoUsuario usuario=${id1} atributo=metodoPagamento
expectError "Usuario nao usa cartao." getAtributoUsuario usuario=${id1} atributo=numeroCartao
expectError "Usuario nao usa pix." getAtributoUsuario usuario=${id1} atributo=chavePix
expectError "Usuario nao usa saldo pre-carregado." getAtributoUsuario usuario=${id1} atributo=saldo

# busca por email
expect ${id1} getUsuarioPorEmail email=joana@exemplo.com
expect ${id2} getUsuarioPorEmail email=loja@exemplo.com
expectError "Usuario nao existe." getUsuarioPorEmail email=ninguem@exemplo.com
expectError "Email nao pode ser nulo." getUsuarioPorEmail email=""

# erros gerais de consulta
expectError "Identificacao do usuario nao pode ser nula." getAtributoUsuario usuario="" atributo=nome
expectError "Usuario nao existe." getAtributoUsuario usuario=U999 atributo=nome
expectError "Atributo nao existe." getAtributoUsuario usuario=${id1} atributo=inexistente
expectError "Atributo nao existe." getAtributoUsuario usuario=${id1} atributo=""

# ---- (refatorado) consulta dos atributos de uma pessoa juridica (cada usuario tem os seus) ----
expect "Loja Bikes ME" getAtributoUsuario usuario=${id2} atributo=nome
expect loja@exemplo.com getAtributoUsuario usuario=${id2} atributo=email
expect 654321 getAtributoUsuario usuario=${id2} atributo=senha
expect ativo getAtributoUsuario usuario=${id2} atributo=estado
expect nenhum getAtributoUsuario usuario=${id2} atributo=aluguelAtivo
expect 0,00 getAtributoUsuario usuario=${id2} atributo=totalPendencias
expect nenhum getAtributoUsuario usuario=${id2} atributo=metodoPagamento

# ---- (refatorado) validade do cartao tambem erra enquanto nao ha metodo de pagamento ----
expectError "Usuario nao usa cartao." getAtributoUsuario usuario=${id1} atributo=validadeCartao
expectError "Usuario nao usa cartao." getAtributoUsuario usuario=${id2} atributo=numeroCartao

# ---- (refatorado) dois usuarios com o mesmo nome: a busca por email distingue ----
id3=criarUsuario nome="Joana Prado" email="joana2@exemplo.com" senha="111111" tipo=pessoaFisica cpf=100.200.300-41
expectDifferent ${id1} echo ${id3}
expect ${id3} getUsuarioPorEmail email=joana2@exemplo.com
expect ${id1} getUsuarioPorEmail email=joana@exemplo.com
expect 111111 getAtributoUsuario usuario=${id3} atributo=senha
expect 123456 getAtributoUsuario usuario=${id1} atributo=senha

# ---- (refatorado) busca por email ignora maiusculas/minusculas ----
expect ${id1} getUsuarioPorEmail email=JOANA@exemplo.com
expect ${id2} getUsuarioPorEmail email=Loja@Exemplo.com

# ---- (refatorado) email so com espacos conta como vazio ----
expectError "Email nao pode ser nulo." getUsuarioPorEmail email="   "

# ---- (refatorado) atributo: so espacos e maiusculas/minusculas nao existem ----
expectError "Atributo nao existe." getAtributoUsuario usuario=${id1} atributo="   "
expectError "Atributo nao existe." getAtributoUsuario usuario=${id1} atributo=Nome
expectError "Atributo nao existe." getAtributoUsuario usuario=${id1} atributo=EMAIL

# ---- (refatorado) ordem das validacoes: usuario e conferido antes do atributo ----
expectError "Identificacao do usuario nao pode ser nula." getAtributoUsuario usuario="" atributo=""
expectError "Usuario nao existe." getAtributoUsuario usuario=U999 atributo=""
expectError "Usuario nao existe." getAtributoUsuario usuario=U999 atributo=inexistente

encerrarSistema
quit
