#####################################################################################
# User Story 15 - Relatorios (disponibilidade, faturamento)
#####################################################################################

zerarSistema

estacao1=criarEstacao nome="Estacao Relatorio 1" endereco="Rua R1, 1" capacidade=4
estacao2=criarEstacao nome="Estacao Relatorio 2" endereco="Rua R2, 2" capacidade=4
adicionarBicicleta estacao=${estacao1} bicicleta=f1 tipo=convencional
adicionarBicicleta estacao=${estacao1} bicicleta=f2 tipo=convencional

usuario1=criarUsuario nome="Camila Duarte" email="camila@exemplo.com" senha="123456" tipo=pessoaFisica cpf=100.300.500-70
alterarUsuario usuario=${usuario1} atributo=metodoPagamento valor1=pix chave=camila@pix.com

# relatorio de disponibilidade (reaproveitando US7)
expect 2 getBicicletasDisponiveis estacao=${estacao1}
expect 2 getVagasDisponiveis estacao=${estacao1}

# corrida 1: finalizada em 5/1/2026, valor 5,00
a1=retirarBicicleta usuario=${usuario1} estacao=${estacao1} bicicleta=f1 data="5/1/2026 08:00"
devolverBicicleta aluguel=${a1} estacao=${estacao2} data="5/1/2026 08:10"

# corrida 2: cancelada em 10/1/2026 (nao entra no faturamento)
a2=retirarBicicleta usuario=${usuario1} estacao=${estacao2} bicicleta=f1 data="10/1/2026 08:00"
devolverBicicleta aluguel=${a2} estacao=${estacao2} data="10/1/2026 08:01"
expect cancelado getAtributoAluguel aluguel=${a2} atributo=estado

# corrida 3: finalizada em 15/1/2026, valor 3,00
a3=retirarBicicleta usuario=${usuario1} estacao=${estacao2} bicicleta=f1 data="15/1/2026 08:00"
devolverBicicleta aluguel=${a3} estacao=${estacao1} data="15/1/2026 08:06"

# corrida 4: finalizada em 1/2/2026, valor 4,00
a4=retirarBicicleta usuario=${usuario1} estacao=${estacao1} bicicleta=f1 data="1/2/2026 08:00"
devolverBicicleta aluguel=${a4} estacao=${estacao2} data="1/2/2026 08:08"

expect 8,00 getTotalFaturado dataInicial=1/1/2026 dataFinal=31/1/2026
expect 12,00 getTotalFaturado dataInicial=1/1/2026 dataFinal=28/2/2026
expect 5,00 getTotalFaturado dataInicial=5/1/2026 dataFinal=5/1/2026
expect 0,00 getTotalFaturado dataInicial=16/1/2026 dataFinal=31/1/2026
expect 0,00 getTotalFaturado dataInicial=1/3/2026 dataFinal=31/3/2026

# ---- erros de validacao ----
expectError "Data nao pode ser nula." getTotalFaturado dataInicial="" dataFinal=31/1/2026
expectError "Data nao pode ser nula." getTotalFaturado dataInicial=1/1/2026 dataFinal=""
expectError "Data invalida." getTotalFaturado dataInicial=31/2/2026 dataFinal=28/2/2026
expectError "Data invalida." getTotalFaturado dataInicial=1/1/2026 dataFinal=abc
expectError "Data final anterior a data inicial." getTotalFaturado dataInicial=31/1/2026 dataFinal=1/1/2026

# ---- (refatorado) datas so com espacos, com horario ou com mes inexistente ----
expectError "Data nao pode ser nula." getTotalFaturado dataInicial="   " dataFinal=31/1/2026
expectError "Data invalida." getTotalFaturado dataInicial="1/1/2026 08:00" dataFinal=31/1/2026
expectError "Data invalida." getTotalFaturado dataInicial=1/13/2026 dataFinal=31/12/2026
# a data inicial e validada antes da final
expectError "Data invalida." getTotalFaturado dataInicial=abc dataFinal=""

# ---- (refatorado) o faturamento conta pelo dia da DEVOLUCAO (quando a cobranca acontece) ----
a5=retirarBicicleta usuario=${usuario1} estacao=${estacao1} bicicleta=f2 data="31/3/2026 23:50"
devolverBicicleta aluguel=${a5} estacao=${estacao2} data="1/4/2026 00:10"
expect 10,00 getAtributoAluguel aluguel=${a5} atributo=valorCobrado
expect 0,00 getTotalFaturado dataInicial=31/3/2026 dataFinal=31/3/2026
expect 10,00 getTotalFaturado dataInicial=1/4/2026 dataFinal=1/4/2026

# ---- (refatorado) devolucao as 23:59 do dia final ainda entra no periodo ----
a6=retirarBicicleta usuario=${usuario1} estacao=${estacao2} bicicleta=f2 data="30/4/2026 23:49"
devolverBicicleta aluguel=${a6} estacao=${estacao1} data="30/4/2026 23:59"
expect 5,00 getTotalFaturado dataInicial=30/4/2026 dataFinal=30/4/2026
expect 15,00 getTotalFaturado dataInicial=1/4/2026 dataFinal=30/4/2026

# ---- (refatorado) aluguel em andamento ainda nao foi cobrado: nao entra ----
a7=retirarBicicleta usuario=${usuario1} estacao=${estacao1} bicicleta=f2 data="5/5/2026 08:00"
expect 0,00 getTotalFaturado dataInicial=1/5/2026 dataFinal=31/5/2026

# ---- (refatorado) varios usuarios, assinante e eletrica (90 min: 30 * 0,30 + 90 * 0,20 = 27,00) ----
usuario2=criarUsuario nome="Heloisa Bento" email="heloisa@exemplo.com" senha="123456" tipo=pessoaFisica cpf=100.300.500-71 \
	plano=assinante mensalidade=50
alterarUsuario usuario=${usuario2} atributo=metodoPagamento valor1=pix chave=heloisa@pix.com
adicionarBicicleta estacao=${estacao1} bicicleta=fE tipo=eletrica nivelBateria=90
a8=retirarBicicleta usuario=${usuario2} estacao=${estacao1} bicicleta=fE data="10/6/2026 08:00"
devolverBicicleta aluguel=${a8} estacao=${estacao2} data="10/6/2026 09:30"
expect 27,00 getAtributoAluguel aluguel=${a8} atributo=valorCobrado
usuario3=criarUsuario nome="Ciclo Entregas" email="ciclo@exemplo.com" senha="123456" \
	tipo=pessoaJuridica cnpj=50.505.050/0001-50 razaoSocial="Ciclo Entregas Ltda"
alterarUsuario usuario=${usuario3} atributo=metodoPagamento valor1=pix chave=ciclo@pix.com
a9=retirarBicicleta usuario=${usuario3} estacao=${estacao2} bicicleta=f1 data="11/6/2026 08:00"
devolverBicicleta aluguel=${a9} estacao=${estacao1} data="11/6/2026 08:04"
expect 29,00 getTotalFaturado dataInicial=1/6/2026 dataFinal=30/6/2026

# ---- (refatorado) total do ano: 12,00 + 15,00 + 29,00 ----
expect 56,00 getTotalFaturado dataInicial=1/1/2026 dataFinal=31/12/2026

encerrarSistema
quit
